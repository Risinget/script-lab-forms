    document.getElementById("insert-data").addEventListener("click", () =>
    tryCatch(insertData));


    async function insertData() {
      console.log("🔍 Iniciando revisión de todas las hojas...");

      await Excel.run(async (context) => {
        const workbook = context.workbook;
        const worksheets = workbook.worksheets;
        worksheets.load("items/name");
        await context.sync();

        let combinedRows = [];
        let totalFilas = 0;

        for (const sheet of worksheets.items) {
          console.log(`📄 Revisando hoja: ${sheet.name}`);

          const cellA1 = sheet.getRange("A1");
          cellA1.load("values");
          await context.sync();

          if (cellA1.values.length > 0 && cellA1.values[0][0] === "N°") {
            console.log(`✅ "${sheet.name}" tiene "N°" en A1.`);

            const usedRange = sheet.getUsedRange();
            usedRange.load("rowCount");
            await context.sync();

            const lastRow = usedRange.rowCount;
            const range = sheet.getRange(`A2:J${lastRow}`);
            range.load("values");
            await context.sync();

            const validRows = range.values.filter((row) => row.some((cell) => cell !== null && cell !== ""));

            console.log(`✔️ ${validRows.length} filas válidas en hoja "${sheet.name}"`);
            totalFilas += validRows.length;
            combinedRows = combinedRows.concat(validRows);
          } else {
            console.log(`❌ "${sheet.name}" no tiene "N°" en A1. Omitiendo...`);
          }
        }

        if (combinedRows.length > 0) {
          const jsonContent = JSON.stringify(combinedRows, null, 2);
          const base64Content = btoa(unescape(encodeURIComponent(jsonContent)));
          const dataUri = `data:application/json;base64,${base64Content}`;

          const downloadLink = document.createElement("a");
          downloadLink.href = dataUri;
          downloadLink.download = "datos-exportados.json";
          downloadLink.textContent = "⬇️ Descargar JSON";
          downloadLink.style.display = "inline-block";
          downloadLink.style.marginTop = "10px";
          downloadLink.style.padding = "8px 12px";
          downloadLink.style.backgroundColor = "#4CAF50";
          downloadLink.style.color = "white";
          downloadLink.style.textDecoration = "none";
          downloadLink.style.borderRadius = "5px";

          const container = document.getElementById("download-link-container");
          if (container) {
            container.innerHTML = "";
            container.appendChild(downloadLink);
          } else {
            document.body.appendChild(downloadLink);
          }

          console.log(`📊 Total de filas válidas en todas las hojas: ${totalFilas}`);
          document.getElementById(
            "status"
          ).innerText = `✅ Total de filas válidas: ${totalFilas} - JSON listo para descargar.`;
        } else {
          console.log("❌ No se encontraron datos válidos.");
          document.getElementById("status").innerText = "❌ No se encontraron datos válidos.";
        }
      });
    }


    async function tryCatch(callback) {
      try {
        await callback();
      } catch (error) {
        console.error("❌ Error:", error);
        document.getElementById("status").innerText = "❌ Error: " + error.message;
      }
    }
