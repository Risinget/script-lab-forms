    document.getElementById("insert-data").addEventListener("click", () =>
    tryCatch(insertData));


    async function insertData() {
      console.log("🔍 Procesando solo la hoja activa...");

      await Excel.run(async (context) => {
        const sheet = context.workbook.worksheets.getActiveWorksheet();

        // Cargar celda A1
        const cellA1 = sheet.getRange("A1");
        cellA1.load("values");
        await context.sync();

        if (cellA1.values.length > 0 && cellA1.values[0][0] === "N°") {
          console.log('✅ A1 contiene "N°". Procesando datos...');

          const usedRange = sheet.getUsedRange();
          usedRange.load("rowCount");
          await context.sync();

          const lastRow = usedRange.rowCount;
          const dataRange = sheet.getRange(`A2:J${lastRow}`);
          dataRange.load("values");
          await context.sync();

          const validRows = dataRange.values.filter((row) => row.some((cell) => cell !== null && cell !== ""));

          console.log(`✅ Total de filas válidas: ${validRows.length}`);
          document.getElementById("status").innerText = `✅ Total de filas válidas: ${validRows.length}`;
          console.log(validRows);
        } else {
          console.log('❌ La celda A1 no contiene "N°".');
          document.getElementById("status").innerText = '❌ La celda A1 no contiene "N°".';
        }
      });
    }


    async function tryCatch(callback) {
      try {
        await callback();
      } catch (error) {
        console.error(error);
        document.getElementById("status").innerText = "❌ Error: " + error.message;
      }
    }
