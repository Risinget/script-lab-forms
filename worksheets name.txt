    document.getElementById("print-sheet-names").addEventListener("click", () =>
    tryCatch(printSheetNames));


    async function printSheetNames() {
      await Excel.run(async (context) => {
        const workbook = context.workbook;
        const worksheets = workbook.worksheets;
        worksheets.load("items/name"); // Carga los nombres de todas las hojas
        await context.sync();

        const sheetNames = worksheets.items.map((sheet) => sheet.name);

        // Imprime los nombres de las hojas en la consola
        console.log("Nombres de las hojas:", sheetNames);

        // Opcional: mostrar los nombres de las hojas en el contenedor HTML
        const container = document.getElementById("sheet-names-container");
        container.innerHTML = ""; // Limpiar contenido previo
        sheetNames.forEach((name) => {
          const paragraph = document.createElement("p");
          paragraph.textContent = name;
          container.appendChild(paragraph);
        });
      });
    }


    async function tryCatch(callback) {
      try {
        await callback();
      } catch (error) {
        console.error(error);
        document.getElementById("status").innerText = "❌ Error: " + error.message;
      }
    }
  language: typescript
